Pin It or Quit It

A fast-paced AR game combining precision, reflexes, and data-driven feedback.

Go to Database Page

Project Overview

Pin It or Quit It is a solo-developed AR game created in eight weeks. The player takes on the role of a crossbow sharpshooter who must react quickly and aim accurately to pin compliments while avoiding insults. Each hit or miss directly affects the player's score, adding pressure as the timer ticks down.

The game was built in Unity (C#) using Unity’s built-in AR Foundation package. It integrates seamlessly with a custom-built web portal and database system that stores session data, scores, and player performance.

Gameplay Mechanics

In Pin It or Quit It, players test their reflexes in a fun and chaotic AR environment. Using a crossbow, you shoot thumbtacks at falling notes, but only the compliments should stick! Every correct hit rewards extra time and points, while mistakes cost you time and score. As the countdown timer ticks, the challenge grows more intense. Can you stay focused and pin all the positivity before time runs out?

Development Process

This project was developed entirely by me in Unity using C#. I integrated Unity’s official AR package to create an interactive, real-world gameplay experience. I designed and implemented all mechanics, from shooting physics to score calculation and database integration. The assets and note designs were inspired by online references, which I adapted and customized to fit the game’s atmosphere. The project also includes a backend connection that sends player data to a web portal for score tracking.

using System.Collections.Generic;
using UnityEngine;

public class Shooting : MonoBehaviour
{
    [SerializeField] private List projectiles;
    private int currentThumbtack = 0;

    [SerializeField] private GameObject shootingPoint;
    private float speed = 600f;

    public bool isHit = false;

    Animator anim;
    AudioSource aud;

    private void Start()
    {
        anim = GetComponent();
        aud = GetComponent();
    }
    void Update()
    {
        if ((Input.GetKeyDown(KeyCode.Space)) || (Input.touchCount > 0 && Input.GetTouch(0).phase == TouchPhase.Began))
        {
            if (anim != null) anim.SetTrigger("Shoot");
            aud.Play();
            Shoot();
        }
    }

    private void Shoot()
    {
        if (projectiles.Count == 0) return;

        GameObject thumbtack = projectiles[currentThumbtack];

        GameObject bullet = Instantiate(thumbtack, shootingPoint.transform.position, shootingPoint.transform.rotation);
        if (bullet != null)
        {
            bullet.GetComponent().AddForce(bullet.transform.forward * speed);

            if (!isHit)
            {
                Destroy(bullet, 2);
            }
        }

        currentThumbtack = (currentThumbtack + 1) % projectiles.Count;

    }

}
                    

public class Score
{
    public static Score Instance { get; private set; }

    public int TotalScore { get; set; }
    public int ComplimentHit { get; set; }
    public int InsultHit { get; set; }

    public static void Reset()
    {
        Instance = new Score();
    }


public class HittingNotes : MonoBehaviour
{
    private void OnTriggerEnter(Collider other)
    {
        if (other != null)
        {
            if (other.CompareTag("Compliment"))
            {
                Score.Instance.ComplimentHit++;
                Score.Instance.TotalScore++;
            }
            else if (other.CompareTag("Insult"))
            {
                Score.Instance.InsultHit++;
                Score.Instance.TotalScore = Score.Instance.TotalScore - 3;  
            }
        }
    }
}

using System;
using System.Collections;
using System.Text;
using UnityEngine;
using UnityEngine.Networking;
using UnityEngine.SceneManagement;

public class CreateSessionApi : MonoBehaviour
{
    private string input;
    private bool usernameExists;
    [SerializeField] private GameObject errorMessage;

    [Serializable]
    private class CreateSessionRequest
    {
        public string username;
    }

    [Serializable]
    private class CreateSessionResponse
    {
        public bool ok;
        public int session_id;
        public string error;
    }

    private const string CreateSessionUrl = "http://YourIPAdress:5173/api/create-session";

    public IEnumerator CreateSession(string username)
    {
        CreateSessionRequest reqObj = new CreateSessionRequest { username = username };
        string json = JsonUtility.ToJson(reqObj);

        UnityWebRequest www = new UnityWebRequest(CreateSessionUrl, "POST");
        www.uploadHandler = new UploadHandlerRaw(Encoding.UTF8.GetBytes(json));
        www.downloadHandler = new DownloadHandlerBuffer();
        www.SetRequestHeader("Content-Type", "application/json");
        www.SetRequestHeader("Accept", "application/json");

        yield return www.SendWebRequest();

        if (www.result == UnityWebRequest.Result.Success)
        {
            string text = www.downloadHandler.text;
            try
            {
                CreateSessionResponse res = JsonUtility.FromJson(text);
                if (res != null && res.ok)
                {
                    Debug.Log($"Session created. session_id = {res.session_id}");
                    GlobalVars.sessionId = res.session_id;
                    usernameExists = true;
                }
                else
                {
                    Debug.LogError($"Server did not return ok. Response: {text}");
                }
            }
            catch (Exception e)
            {
                Debug.LogError($"Failed to parse JSON: {e.Message}\nResponse: {text}");
            }
        }
        else
        {
            Debug.LogError($"HTTP error: {www.responseCode} - {www.error}");
            try
            {
                string text = www.downloadHandler.text;
                CreateSessionResponse res = JsonUtility.FromJson(text);
                if (res != null && !string.IsNullOrEmpty(res.error))
                {
                    Debug.LogError($"The server provided the following error message: {res.error}");
                }
            }
            catch (Exception e)
            {
                Debug.LogError($"Failed to parse JSON: {e.Message}");
            }
        }
    }

    public void CreateSession()
    {
        StartCoroutine(CreateTheSessionAsync());
    }
    private IEnumerator CreateTheSessionAsync()
    {
        yield return StartCoroutine(CreateSession(input));
        if (usernameExists)
        {
            SceneManager.LoadScene("Main scene");
        }
        else
        {
            errorMessage.SetActive(true);
            yield return new WaitForSeconds(5);
            errorMessage.SetActive(false);
        }
    }

    public void ReadInputField(string s)
    {
        input = s;
        Debug.Log(input);
        GlobalVars.username = input;
    }
}

        www.SetRequestHeader("Content-Type", "application/json");
        www.SetRequestHeader("Accept", "application/json");

        yield return www.SendWebRequest();
        Debug.Log(www.result + " www");

        {
            Debug.Log(www.result + " www");

            {
                SubmitScoreResponse res = JsonUtility.FromJson(text);

                }
                else
                {

            }

        }

            }
            catch (Exception e)
            {
                Debug.LogError($"Failed to parse JSON: {e.Message}");
            }
        }
    }

    private void Start()
    {
        gameTimer = FindFirstObjectByType();
        timePlayed = FindFirstObjectByType();
        speedLevel = FindFirstObjectByType();
    }

    private void Update()
    {
        TimesUp();
    }
    private void TimesUp()
    {
        if (gameTimer.TimeLeft <= 0f)
        {
            int sessionId = GlobalVars.sessionId;
            int levelId = 1;
            Debug.Log(sessionId);

            StartCoroutine(SubmitScore(sessionId, levelId, timePlayed.TimePlaying, speedLevel.speed));
        }

Challenges & Learnings

Developing Pin It or Quit It entirely on my own taught me valuable lessons about full-cycle development. One major challenge was synchronizing real-time AR gameplay with server-side data saving without interrupting performance. I learned how to handle asynchronous API calls in Unity, manage time-based gameplay mechanics, and implement visual feedback for user interaction.

Additionally, balancing intuitive design with a technically stable backend pushed me to think critically about architecture and user experience. The result was a fully functional, data-driven AR game that I’m proud to have created from start to finish.
